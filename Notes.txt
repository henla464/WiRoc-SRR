from smbus2 import SMBus, i2c_msg
with SMBus(0) as bus:
    a = 0x20
    status = bus.read_byte_data(a, 0x04)                # bit7=error pending, bit0=punch pending
    c = bus.read_i2c_block_data(a, 0x03, 2)             # error count (uint16 LE)
    count = c[0] | (c[1] << 8)
    length = bus.read_byte_data(a, 0x27)                # length of last error string
    if length:
        w = i2c_msg.write(a, [0x47])                    # select ERRORMSG register
        r = i2c_msg.read(a, length)                     # raw read — no 32-byte cap
        bus.i2c_rdwr(w, r)
        msg = bytes(r)
    else:
        msg = b""
print("status=0x%02x  errors=%d  len=%d" % (status, count, length))
print("message:", msg.decode("latin1"))
'

 #!/usr/bin/env python3
from smbus2 import SMBus

bus = SMBus(0)
addr = 0x20

def read_error_msg():
    n = bus.read_byte_data(addr, 0x27)          # ERRORLENGTH
    if n == 0:
        return ""
    buf = bytearray()
    idx = 0
    while idx < n:
        bus.write_byte_data(addr, 0x05, idx)    # SETDATAINDEX
        chunk = min(n - idx, 31)                # SMBus block limit
        buf += bytearray(bus.read_i2c_block_data(addr, 0x47, chunk))  # ERRORMSG
        idx += chunk
    return bytes(buf).decode(errors="replace")

count = bus.read_word_data(addr, 0x03)          # ERRORCOUNT (uint16, big→little handled by smb
print("log count:", count)
print("last:", read_error_msg())
